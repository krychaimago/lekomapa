// Prototype: ten desktop concepts. Every concept gives each arrival (field f, in order) an anchor point from f alone,
// then a scatter from the id; a button that would hit one already standing tries smaller scatter, then the nearest
// free spot. Places never depend on interactions and never change once taken. Not app code.
const BOX_W = 90, BOX_H = 88, TOP = 10 / 0.82, GAP = 4;
export function hash(s, salt) {
  let h = 2166136261 ^ salt;
  for (let k = 0; k < s.length; k++) {
    h ^= s.charCodeAt(k);
    h = Math.imul(h, 16777619);
  }
  h ^= h >>> 13;
  h = Math.imul(h, 0x5bd1e995);
  h ^= h >>> 15;
  return (h >>> 0) / 4294967295;
}
export function place(meds, { width, anchor, jx, jy }) {
  const box = (p) => ({ l: p.x - BOX_W / 2, r: p.x + BOX_W / 2, t: p.y - TOP, b: p.y - TOP + BOX_H });
  const hits = (p, placed) =>
    placed.some((q) => {
      const a = box(p), b = box(q);
      return a.l < b.r && b.l < a.r && a.t < b.b + GAP && b.t < a.b + GAP;
    });
  const clampX = (x) => Math.max(BOX_W / 2 + 4, Math.min(width - BOX_W / 2 - 4, x));
  const placed = [];
  meds.forEach((m, f) => {
    const { x: cx, y: cy } = anchor(f);
    let p = null;
    for (let i = 0; i < 24 && !p; i++) {
      const s = 1 - i / 24;
      const c = { x: clampX(cx + (hash(m.id, 2 + 2 * i) - 0.5) * 2 * jx * s), y: Math.max(26, cy + (hash(m.id, 3 + 2 * i) - 0.5) * 2 * jy * s) };
      if (!hits(c, placed)) p = c;
    }
    for (let r = 6; !p; r += 6)
      for (let a = 0; a < 360 && !p; a += 15) {
        const c = { x: clampX(cx + r * Math.cos((a * Math.PI) / 180)), y: Math.max(26, cy + r * Math.sin((a * Math.PI) / 180)) };
        if (!hits(c, placed)) p = c;
      }
    placed.push({ ...m, ...p });
  });
  const height = Math.max(...placed.map((p) => p.y - TOP + BOX_H)) + 12;
  return { width, height, nodes: placed };
}
const W = 576, CW = 108;
const golden = Math.PI * (3 - Math.sqrt(5));
// Field f → cell of a block, scattered by a stride.
const cellOf = (f, cols, rows, stride) => {
  const block = cols * rows, b = Math.floor(f / block), cell = (f * stride) % block;
  return { col: cell % cols, row: rows * b + Math.floor(cell / cols), cell };
};
