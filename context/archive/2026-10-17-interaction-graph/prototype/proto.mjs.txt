// Prototype: free placement like mockup A2, avoiding already placed medications. Not app code.
const CW = 108, BOX_W = 90, BOX_H = 88, TOP = 10 / 0.82, GAP = 4;
function hash(s, salt) {
  let h = 2166136261 ^ salt;
  for (let k = 0; k < s.length; k++) { h ^= s.charCodeAt(k); h = Math.imul(h, 16777619); }
  h ^= h >>> 13; h = Math.imul(h, 0x5bd1e995); h ^= h >>> 15;
  return (h >>> 0) / 4294967295;
}
export function layout(meds, cols, { ch = 112, jx = 0.3, jy = 0.3, rows = 4, stride = 7, offsets = [0, 0, 0], cellShift = 0 } = {}) {
  // meds: [{id, name}] in arrival order (all current; enough for the mockup).
  const width = cols * CW + 36, block = cols * rows;
  const box = (p) => ({ l: p.x - BOX_W / 2, r: p.x + BOX_W / 2, t: p.y - TOP, b: p.y - TOP + BOX_H });
  const hits = (p, placed) => placed.some((q) => { const a = box(p), b = box(q); return a.l < b.r && b.l < a.r && a.t < b.b + GAP && b.t < a.b + GAP; });
  const clampX = (x) => Math.max(BOX_W / 2 + 4, Math.min(width - BOX_W / 2 - 4, x));
  const placed = [];
  meds.forEach((m, f) => {
    const b = Math.floor(f / block), cell = (f * stride) % block;
    const col = cell % cols, row = rows * b + Math.floor(cell / cols);
    const shift = row % 2 ? CW * 0.28 : -CW * 0.12;
    // Columns starting at different heights, and every field moved up or down by its own number, break the rows.
    const lift = ch * (offsets[col] + cellShift * (((cell * 5 + 3) % 7) / 6 - 0.5));
    let p = null;
    for (let i = 0; i < 24 && !p; i++) {
      const scale = 1 - i / 24;
      const c = { x: clampX(18 + CW * (col + 0.5) + shift * scale + (hash(m.id, 2 + 2 * i) - 0.5) * 2 * jx * CW * scale),
                  y: Math.max(26, 26 + lift + ch * row + ch * 0.3 + (hash(m.id, 3 + 2 * i) - 0.5) * 2 * jy * ch * scale) };
      if (!hits(c, placed)) p = c;
    }
    // Nearest free spot to the field's centre, in growing rings, instead of the first free spot scanning down.
    const fx = clampX(18 + CW * (col + 0.5)), fy = 26 + lift + ch * row + ch * 0.3;
    for (let r = 6; !p; r += 6) for (let a = 0; a < 360 && !p; a += 15) {
      const c = { x: clampX(fx + r * Math.cos((a * Math.PI) / 180)), y: Math.max(26, fy + r * Math.sin((a * Math.PI) / 180)) };
      if (!hits(c, placed)) p = c;
    }
    placed.push({ ...m, ...p });
  });
  const height = Math.max(...placed.map((p) => p.y - TOP + BOX_H)) + 12;
  return { width, height, nodes: placed };
}
